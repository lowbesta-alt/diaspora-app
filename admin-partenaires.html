async function saveEdit() { const id = document.getElementById('edit-id').value; const company = document.getElementById('edit-company').value.trim(); const newEmail = document.getElementById('edit-email').value.trim().toLowerCase(); if (!company) { alert('Le nom de l\'entreprise est obligatoire'); return; } if (!newEmail) { alert('L\'email est obligatoire'); return; } const originalPartner = allPartners.find(p => p.id === id); const oldEmail = originalPartner ? (originalPartner.email || '').toLowerCase().trim() : ''; const emailChanged = oldEmail && oldEmail !== newEmail; const btn = document.getElementById('saveEditBtn'); btn.disabled = true; btn.textContent = 'Enregistrement...'; try { await apiPatch('partnership_requests', id, { company_name: company, profession: document.getElementById('edit-profession').value.trim() || null, contact_name: document.getElementById('edit-contact').value.trim() || null, phone: document.getElementById('edit-phone').value.trim() || null, city: document.getElementById('edit-city').value.trim() || null, country: document.getElementById('edit-country').value.trim() || null, description: document.getElementById('edit-description').value.trim() || null }); if (emailChanged) { try { const res = await fetch(SUPABASE_URL + '/rest/v1/rpc/admin_update_partner_email', { method: 'POST', headers: { 'apikey': SUPABASE_KEY, 'Authorization': 'Bearer ' + (getToken() || SUPABASE_KEY), 'Content-Type': 'application/json' }, body: JSON.stringify({ p_partnership_id: id, p_new_email: newEmail }) }); if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.message || 'Erreur de mise a jour email'); } const data = await res.json(); if (!data.success) throw new Error('RPC a echoue'); showSuccess('Email mis a jour partout !'); } catch (err) { alert('Attention : candidature modifiee, MAIS email non propage. Erreur : ' + err.message); closeEditModal(); await loadPartners(); btn.disabled = false; btn.textContent = 'Enregistrer'; return; } } else { await apiPatch('partnership_requests', id, { email: newEmail }); showSuccess('Candidature modifiee !'); } try { await apiPost('audit_log', { actor_id: getCurrentUserId(), action: 'edit_partner', table_name: 'partnership_requests', record_id: id, new_data: { company_name: company, email: newEmail, email_changed: emailChanged } }); } catch (e) {} closeEditModal(); await loadPartners(); } catch (e) { alert('Erreur : ' + e.message); } btn.disabled = false; btn.textContent = 'Enregistrer'; }
